import re, pathlib, collections
patterns = {
 'advanceView': r'advanceView',
 'HighQC updated': r'Successfully updated HighQC',

        for i, line in enumerate(lines):
            if re.search(pat, line):
                # search in window [i-3, i+3]
                found = None
